local ESPEnabled = false
local ESPObjects = {}

local function RemoveESP(Player)
    if ESPObjects[Player] then
        for _, Object in ipairs(ESPObjects[Player]) do
            if Object and Object.Parent then
                Object:Destroy()
            end
        end
        ESPObjects[Player] = nil
    end
end

local function AddESP(Player)
    if not Player or Player.Name ~= "CriminalBlueGuy" then
        return
    end

    RemoveESP(Player)

    local Character = Player.Character
    if not Character then
        return
    end

    local Highlight = Instance.new("Highlight")
    Highlight.Name = "SlopwareESP"
    Highlight.Adornee = Character
    Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    Highlight.FillTransparency = 0.5
    Highlight.OutlineTransparency = 0
    Highlight.Parent = Character

    local Billboard = Instance.new("BillboardGui")
    Billboard.Name = "SlopwareESPText"
    Billboard.Size = UDim2.fromOffset(150, 40)
    Billboard.StudsOffset = Vector3.new(0, 3.5, 0)
    Billboard.AlwaysOnTop = true
    Billboard.Parent = Character

    local Text = Instance.new("TextLabel")
    Text.Size = UDim2.fromScale(1, 1)
    Text.BackgroundTransparency = 1
    Text.Text = "right here"
    Text.TextScaled = true
    Text.Font = Enum.Font.GothamBold
    Text.TextColor3 = Color3.new(1, 1, 1)
    Text.TextStrokeTransparency = 0
    Text.Parent = Billboard

    ESPObjects[Player] = {
        Highlight,
        Billboard
    }
end

local function WatchESPPlayer(Player)
    Player.CharacterAdded:Connect(function()
        task.wait(0.5)

        if ESPEnabled then
            AddESP(Player)
        end
    end)

    if ESPEnabled and Player.Character then
        AddESP(Player)
    end
end

local UserHideEnabled = false

local function ScrambleUserText()
    local Chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
    local Result = ""

    for _ = 1, math.random(6, 14) do
        local Index = math.random(1, #Chars)
        Result ..= Chars:sub(Index, Index)
    end

    return Result
end

local function FindLocalPlayerScoreLabels(Scoreboard)
    if not Scoreboard then
        return
    end

    for _, Object in ipairs(Scoreboard:GetDescendants()) do
        if Object:IsA("TextLabel") then
            if Object.Text == LocalPlayer.Name
                or Object.Text == LocalPlayer.DisplayName then
                ScrambledScoreLabels[Object] = true
                Object.Text = ScrambleUserText()
            end
        end
    end
end

local function HideUserName()
    if not UserHideEnabled then
        return
    end

    local Character = LocalPlayer.Character
    local Head = Character and Character:FindFirstChild("Head")
    local NameTag = Head and Head:FindFirstChild("NameTag")
    local TextLabel = NameTag and NameTag:FindFirstChild("TextLabel")

    if TextLabel and TextLabel:IsA("TextLabel") then
        TextLabel.Text = ScrambleUserText()
    end

    local Scoreboard = GetScoreboardFolder()

    if Scoreboard then
        FindLocalPlayerScoreLabels(Scoreboard)

        for Object in pairs(ScrambledScoreLabels) do
            if Object and Object.Parent and Object:IsDescendantOf(Scoreboard) then
                Object.Text = ScrambleUserText()
            else
                ScrambledScoreLabels[Object] = nil
            end
        end
    else
        table.clear(ScrambledScoreLabels)
    end
end

local function GardenPartIsOwnedByLocalPlayer(GardenPart)
    local Owner = GardenPart:FindFirstChild("PartOwner", true)

    if not Owner then
        return false
    end

    local OwnerValue

    pcall(function()
        OwnerValue = Owner.Value
    end)

    if typeof(OwnerValue) == "Instance" then
        return OwnerValue == LocalPlayer
    end

    if typeof(OwnerValue) == "number" then
        return OwnerValue == LocalPlayer.UserId
    end

    if typeof(OwnerValue) == "string" then
        return OwnerValue == LocalPlayer.Name
            or OwnerValue == LocalPlayer.DisplayName
            or OwnerValue == tostring(LocalPlayer.UserId)
    end

    return false
end

local AutoShoot = false
local CurrentTarget = nil
local TargetStartTime = 0
local LastTarget = nil

local function GetRandomTarget()
    local Targets = {}

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player ~= LastTarget then
            local Character = Player.Character
            local Humanoid =
                Character
                and Character:FindFirstChildOfClass("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                table.insert(Targets, Player)
            end
        end
    end

    if #Targets == 0 then
        for _, Player in ipairs(Players:GetPlayers()) do
            if Player ~= LocalPlayer then
                local Character = Player.Character
                local Humanoid =
                    Character
                    and Character:FindFirstChildOfClass("Humanoid")

                if Humanoid and Humanoid.Health > 0 then
                    table.insert(Targets, Player)
                end
            end
        end
    end

    if #Targets > 0 then
        return Targets[math.random(1, #Targets)]
    end

    return nil
end

local function ShootTarget(TargetPlayer)
    if not TargetPlayer then
        return false
    end

    local Character = LocalPlayer.Character

    if not Character then
        return false
    end

    local Root =
        Character:FindFirstChild("HumanoidRootPart")

    local Pistol =
        Character:FindFirstChild("PistolTool")

    local Activation =
        Pistol
        and Pistol:FindFirstChild("Activation")

    if not Root or not Activation then
        return false
    end

    local TargetCharacter = TargetPlayer.Character

    if not TargetCharacter then
        return false
    end

    local TargetRoot =
        TargetCharacter:FindFirstChild("HumanoidRootPart")

    local Humanoid =
        TargetCharacter:FindFirstChildOfClass("Humanoid")

    if not TargetRoot
        or not Humanoid
        or Humanoid.Health <= 0 then
        return false
    end

    Root.CFrame =
        TargetRoot.CFrame
        * CFrame.new(0, 0, 5)

    pcall(function()
        Activation:FireServer(
            TargetRoot,
            TargetRoot.CFrame
        )
    end)

    return true
end

local function AutoWinShoot()
    if not CurrentTarget
        or not CurrentTarget.Character
        or not CurrentTarget.Character:FindFirstChildOfClass("Humanoid")
        or CurrentTarget.Character:FindFirstChildOfClass("Humanoid").Health <= 0
        or tick() - TargetStartTime >= 2 then

        LastTarget = CurrentTarget
        CurrentTarget = GetRandomTarget()
        TargetStartTime = tick()
    end

    if CurrentTarget then
        if not ShootTarget(CurrentTarget) then
            LastTarget = CurrentTarget
            CurrentTarget = GetRandomTarget()
            TargetStartTime = tick()
        end
    end
end

            end

            if CurrentTarget then
                if not ShootTarget(CurrentTarget) then

                    TargetStartTime = tick()

        end
    end
end)

local function GetLocalPlayerMap()
    local GameSystem =
        workspace:FindFirstChild("GameSystem", true)

    if not GameSystem then
        return nil
    end

    local DeleteAfterMinigames = {}

    for _, Object in ipairs(GameSystem:GetDescendants()) do
        if Object.Name == "DeleteAfterMinigame" then
            table.insert(DeleteAfterMinigames, Object)
        end
    end

    local DirectDeleteAfterMinigame =
        GameSystem:FindFirstChild("DeleteAfterMinigame")

    if DirectDeleteAfterMinigame then
        table.insert(DeleteAfterMinigames, DirectDeleteAfterMinigame)
    end

    local ExpectedName = LocalPlayer.Name .. "Map"

    for _, DeleteAfterMinigame in ipairs(DeleteAfterMinigames) do
        for _, Object in ipairs(DeleteAfterMinigame:GetDescendants()) do
            if Object.Name == "Map" then
                local PlayerMap =
                    Object:FindFirstChild(ExpectedName)

                if PlayerMap then
                    return PlayerMap
                end
            end

            if Object.Name == ExpectedName then
                return Object
            end
        end
    end

    return nil
end

local function GetObjectCFrame(Object)
    if not Object or not Object.Parent then
        return nil
    end

    if Object:IsA("BasePart") then
        return Object.CFrame
    end

    if Object:IsA("Model") then
        return Object:GetPivot()
    end

    if Object:IsA("Attachment") then
        return Object.WorldCFrame
    end

    local Part = Object:FindFirstChildWhichIsA("BasePart", true)

    if Part then
        return Part.CFrame
    end

    return nil
end

local function MoveSlowlyToObject(Object, TouchGate)
    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local TargetCFrame = GetObjectCFrame(Object)

    if not Root or not TargetCFrame then
        return false
    end

    if not TouchGate then
        TargetCFrame = TargetCFrame + Vector3.new(0, 3, 5)
    end

    local Distance =
        (Root.Position - TargetCFrame.Position).Magnitude

    local MoveTime =
        math.clamp(Distance / 35, 0.5, 2.5)

    local Tween = TweenService:Create(
        Root,
        TweenInfo.new(
            MoveTime,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        ),
        {CFrame = TargetCFrame}
    )

    Tween:Play()

    while Tween.PlaybackState == Enum.PlaybackState.Playing do
        if not AutoDetectWin then
            Tween:Cancel()
            return false
        end

        task.wait()
    end

    return true
end

local function TeleportToObject(Object)
    return MoveSlowlyToObject(Object, false)
end

local function ExpandEnemyHitbox(Enemy)
    if not Enemy or not Enemy.Parent then
        return
    end

    if not EnemyNames[Enemy.Name] then
        return
    end

    local Root =
        Enemy:FindFirstChild("HumanoidRootPart", true)

    if Root and Root:IsA("BasePart") then
        pcall(function()
            Root.Size = EnemyHitboxSize
            Root.Transparency = 0.8
            Root.CanCollide = false
            Root.CanTouch = true
            Root.CanQuery = true
        end)
    end
end

local function WatchEnemyFolder(Enemies)
    if not Enemies then
        return
    end

    for _, Enemy in ipairs(Enemies:GetChildren()) do
        ExpandEnemyHitbox(Enemy)
    end

    if WatchedEnemyFolders[Enemies] then
        return
    end

    WatchedEnemyFolders[Enemies] = true

    Enemies.ChildAdded:Connect(function(Enemy)
        task.wait()

        if AutoDetectWin then
            ExpandEnemyHitbox(Enemy)
        end
    end)
end

local SwordFightHitboxSize = Vector3.new(500, 500, 500)
local SwordFightOriginalHitboxes = {}
local SwordFightLastParrying = false

local function IsPlayerParrying(Player)
    if not Player or Player == LocalPlayer then
        return false
    end

    local Character = Player.Character

    if not Character then
        return false
    end

    local IsParrying =
        Character:FindFirstChild("IsParrying", true)

    if not IsParrying then
        return false
    end

    local Value

    pcall(function()
        Value = IsParrying.Value
    end)

    return Value == true
end

local function ExpandSwordFightHitbox(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    local Character = Player.Character

    if not Character then
        return
    end

    local Root =
        Character:FindFirstChild("HumanoidRootPart")

    if not Root or not Root:IsA("BasePart") then
        return
    end

    if not SwordFightOriginalHitboxes[Player] then
        SwordFightOriginalHitboxes[Player] = {
            Size = Root.Size,
            Transparency = Root.Transparency,
            CanCollide = Root.CanCollide,
            CanTouch = Root.CanTouch,
            CanQuery = Root.CanQuery
        }
    end

    pcall(function()
        Root.Size = SwordFightHitboxSize
        Root.Transparency = 0.8
        Root.CanCollide = false
        Root.CanTouch = true
        Root.CanQuery = true
    end)
end

local function ExpandAllSwordFightHitboxes()
    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer then
            ExpandSwordFightHitbox(Player)
        end
    end
end

local function RestoreSwordFightHitboxes()
    for Player, Original in pairs(SwordFightOriginalHitboxes) do
        local Character = Player and Player.Character
        local Root =
            Character
            and Character:FindFirstChild("HumanoidRootPart")

        if Root and Original then
            pcall(function()
                Root.Size = Original.Size
                Root.Transparency = Original.Transparency
                Root.CanCollide = Original.CanCollide
                Root.CanTouch = Original.CanTouch
                Root.CanQuery = Original.CanQuery
            end)
        end
    end

    table.clear(SwordFightOriginalHitboxes)
end

local function TeleportToAutoWinSafeSpot()
    CreateAutoWinBaseplate()
end

local function AttackSwordFightPlayer(Player)
    if not Player or Player == LocalPlayer then
        return
    end

    if IsPlayerParrying(Player) then
        return
    end

    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local TargetCharacter = Player.Character
    local TargetRoot =
        TargetCharacter
        and TargetCharacter:FindFirstChild("HumanoidRootPart")

    local Humanoid =
        TargetCharacter
        and TargetCharacter:FindFirstChildOfClass("Humanoid")

    if not Root or not TargetRoot or not Humanoid or Humanoid.Health <= 0 then
        return
    end

    ExpandSwordFightHitbox(Player)

    local TargetCFrame =
        TargetRoot.CFrame
        * CFrame.new(0, -13, 0)

    pcall(function()
        Character:PivotTo(TargetCFrame)
        Root.CFrame = TargetCFrame
        Root.AssemblyLinearVelocity = Vector3.zero
        Root.AssemblyAngularVelocity = Vector3.zero
    end)

    local Event = GetSwordEvent()

    if Event then
        for _ = 1, 6 do
            if not AutoDetectWin or not TargetRoot.Parent then
                break
            end

            pcall(function()
                Character:PivotTo(TargetCFrame)
                Root.CFrame = TargetCFrame
                Root.AssemblyLinearVelocity = Vector3.zero
                Root.AssemblyAngularVelocity = Vector3.zero
                Event:FireServer()
            end)

            task.wait(0.05)
        end
    end
end

local function AutoWinSwordFight()
    ExpandAllSwordFightHitboxes()

    local Event = GetSwordEvent()

    if not Event then
        return
    end

    local ParryingPlayerFound = false

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer
            and Player.Character then

            local Humanoid =
                Player.Character:FindFirstChildOfClass("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                if IsPlayerParrying(Player) then
                    ParryingPlayerFound = true
                end
            end
        end
    end

    if ParryingPlayerFound then
        if not SwordFightLastParrying then
            SwordFightLastParrying = true
        end

        TeleportToAutoWinSafeSpot()
        return
    end

    SwordFightLastParrying = false

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer
            and Player.Character
            and not IsPlayerParrying(Player) then

            local Humanoid =
                Player.Character:FindFirstChildOfClass("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                AttackSwordFightPlayer(Player)
            end
        end
    end
end

local function TeleportUnderEnemy(Enemy)
    if not Enemy or not Enemy.Parent then
        return false
    end

    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local EnemyRoot =
        Enemy:FindFirstChild("HumanoidRootPart", true)

    if not Root or not EnemyRoot or not EnemyRoot:IsA("BasePart") then
        return false
    end

    -- Instantly place the player 15 studs UNDER the zombie.
    -- This is deliberately below the enemy, not on top of the map.
    pcall(function()
        Root.CFrame =
            EnemyRoot.CFrame
            * CFrame.new(0, -13, 0)
    end)

    return true
end

local function AttackEnemy(Enemy)
    if not Enemy or not Enemy.Parent then
        return false
    end

    if not EnemyNames[Enemy.Name] then
        return false
    end

    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    if not Root then
        return false
    end

    local EnemyRoot =
        Enemy:FindFirstChild("HumanoidRootPart", true)

    if not EnemyRoot and Enemy:IsA("BasePart") then
        EnemyRoot = Enemy
    end

    if not EnemyRoot then
        return false
    end

    ExpandEnemyHitbox(Enemy)

    -- Instantly place the player on the real ground under the spawned enemy.
    -- Do this before attacking so the slow sword loop cannot leave the player exposed.
    TeleportUnderEnemy(Enemy)

    local Event = GetSwordEvent()

    if not Event then
        return false
    end

    for _ = 1, 10 do
        if not AutoDetectWin or not Enemy.Parent then
            break
        end

        local Humanoid =
            Enemy:FindFirstChildOfClass("Humanoid")

        if Humanoid and Humanoid.Health <= 0 then
            break
        end

        pcall(function()
            Event:FireServer()
        end)

        task.wait(0.15)
    end

    return true
end

local function GetTileNumber(Tile)
    local Number =
        string.match(Tile.Name, "^Tile(%d+)$")

    return tonumber(Number) or math.huge
end

local function GetTiles(UserMap)
    local Tiles = {}

    for _, Object in ipairs(UserMap:GetChildren()) do
        if (Object:IsA("Model") or Object:IsA("Folder"))
            and string.match(Object.Name, "^Tile%d+$") then
            table.insert(Tiles, Object)
        end
    end

    table.sort(Tiles, function(A, B)
        local ANumber = GetTileNumber(A)
        local BNumber = GetTileNumber(B)

        if ANumber == BNumber then
            return A.Name < B.Name
        end

        return ANumber < BNumber
    end)

    return Tiles
end

local function GetOpenGate(UserMapOrTile)
    local StartPath =
        UserMapOrTile:FindFirstChild("StartPath", true)

    if StartPath then
        local Gate =
            StartPath:FindFirstChild("OpenGate", true)

        if Gate then
            return Gate
        end
    end

    return UserMapOrTile:FindFirstChild("OpenGate", true)
end

local function GetCloseGate(Tile)
    return Tile:FindFirstChild("CloseGate", true)
end

local function GetEnemyFolder(Tile)
    return Tile:FindFirstChild("Enemies", true)
end

local function GetEnemiesLeftValue(Tile)
    local EnemiesLeft =
        Tile:FindFirstChild("EnemiesLeft", true)

    if not EnemiesLeft then
        return nil
    end

    local Value

    pcall(function()
        Value = EnemiesLeft.Value
    end)

    if typeof(Value) == "number" then
        return Value
    end

    return nil
end

local function GetAliveEnemies(Enemies)
    local Alive = {}
    local Seen = {}

    if not Enemies then
        return Alive
    end

    for _, Enemy in ipairs(Enemies:GetDescendants()) do
        if EnemyNames[Enemy.Name] and not Seen[Enemy] then
            local Humanoid =
                Enemy:FindFirstChildOfClass("Humanoid")
                or Enemy:FindFirstChild("Humanoid", true)

            if Humanoid and Humanoid.Health > 0 then
                Seen[Enemy] = true
                table.insert(Alive, Enemy)
            end
        end
    end

    for _, Enemy in ipairs(Enemies:GetChildren()) do
        if EnemyNames[Enemy.Name] and not Seen[Enemy] then
            local Humanoid =
                Enemy:FindFirstChildOfClass("Humanoid")
                or Enemy:FindFirstChild("Humanoid", true)

            if Humanoid and Humanoid.Health > 0 then
                Seen[Enemy] = true
                table.insert(Alive, Enemy)
            end
        end
    end

    return Alive
end

local function WaitForTileEnemiesToFinish(Tile, Enemies)
    local Spawned = false
    local EmptyChecks = 0

    while AutoDetectWin and Tile.Parent and Enemies.Parent do
        local Alive = GetAliveEnemies(Enemies)

        if #Alive > 0 then
            Spawned = true
            EmptyChecks = 0

            for _, Enemy in ipairs(Alive) do
                if not AutoDetectWin then
                    return false
                end

                if Enemy.Parent then
                    local Humanoid =
                        Enemy:FindFirstChildOfClass("Humanoid")
                        or Enemy:FindFirstChild("Humanoid", true)

                    if Humanoid and Humanoid.Health > 0 then
                        ExpandEnemyHitbox(Enemy)
                        TeleportUnderEnemy(Enemy)
                        AttackEnemy(Enemy)
                    end
                end
            end

            task.wait(0.08)
        elseif not Spawned then
            task.wait(0.1)
        else
            EmptyChecks += 1

            if EmptyChecks >= 5 then
                return true
            end

            task.wait(0.1)
        end
    end

    return false
end

local function MoveToNextTile(UserMap, ProcessedTiles)
    local Tiles = GetTiles(UserMap)
    local NextTile

    for _, Candidate in ipairs(Tiles) do
        if not ProcessedTiles[Candidate] then
            NextTile = Candidate
            break
        end
    end

    if not NextTile then
        return true
    end

    local NextOpenGate = GetOpenGate(NextTile)

    if not NextOpenGate then
        return true
    end

    for _ = 1, 3 do
        if not AutoDetectWin then
            return false
        end

        if MoveSlowlyToObject(NextOpenGate, false) then
            LastPlayerMapGate = NextOpenGate
            task.wait(0.2)
            return true
        end

        task.wait(0.15)
    end

    return false
end

local function HandleTile(Tile)
    if not Tile or not Tile.Parent then
        return false
    end

    local OpenGate = GetOpenGate(Tile)

    if OpenGate then
        local Reached = false

        for _ = 1, 3 do
            if not AutoDetectWin then
                return false
            end

            if MoveSlowlyToObject(OpenGate, false) then
                Reached = true
                break
            end

            task.wait(0.15)
        end

        if not Reached then
            return false
        end

        LastPlayerMapGate = OpenGate
        task.wait(0.15)
    end

    local CloseGate = GetCloseGate(Tile)

    if CloseGate then
        local Reached = false

        for _ = 1, 3 do
            if not AutoDetectWin then
                return false
            end

            if MoveSlowlyToObject(CloseGate, true) then
                Reached = true
                break
            end

            task.wait(0.15)
        end

        if not Reached then
            return false
        end

        LastPlayerMapGate = CloseGate
        task.wait(0.15)
    end

    local Enemies = GetEnemyFolder(Tile)

    while AutoDetectWin and Tile.Parent and not Enemies do
        Enemies = GetEnemyFolder(Tile)
        task.wait(0.1)
    end

    if not Enemies then
        return false
    end

    WatchEnemyFolder(Enemies)

    return WaitForTileEnemiesToFinish(Tile, Enemies)
end

local function AutoWinPlayerMap()
    local UserMap = GetLocalPlayerMap()

    if not UserMap then
        return false
    end

    if not PlayerMapStarted then
        local StartPath =
            UserMap:FindFirstChild("StartPath", true)

        local OpenGate =
            StartPath
            and StartPath:FindFirstChild("OpenGate", true)

        if not OpenGate then
            OpenGate =
                UserMap:FindFirstChild("OpenGate", true)
        end

        if OpenGate then
            local Reached = false

            for _ = 1, 3 do
                if not AutoDetectWin then
                    return false
                end

                if MoveSlowlyToObject(OpenGate, false) then
                    Reached = true
                    break
                end

                task.wait(0.15)
            end

            if not Reached then
                return false
            end

            PlayerMapStarted = true
            LastPlayerMapGate = OpenGate
            task.wait(0.25)
        end
    end

    local ProcessedTiles = {}

    while AutoDetectWin and UserMap.Parent do
        local Tiles = GetTiles(UserMap)
        local DidWork = false

        for _, Tile in ipairs(Tiles) do
            if not AutoDetectWin then
                return false
            end

            if not ProcessedTiles[Tile] then
                LastPlayerMapTile = Tile

                local Success = HandleTile(Tile)

                if not Success then
                    task.wait(0.2)
                    break
                end

                ProcessedTiles[Tile] = true
                DidWork = true

                if not MoveToNextTile(UserMap, ProcessedTiles) then
                    task.wait(0.2)
                    break
                end

                task.wait(0.1)
            end
        end

        if not DidWork then
            task.wait(0.1)
        end
    end

    return true
end

local HitboxEnabled = false
local OriginalHitboxes = {}

local function SetHitbox(Player)
    if Player == LocalPlayer then
        return
    end

    local Character = Player.Character

    if not Character then
        return
    end

    local Root =
        Character:FindFirstChild("HumanoidRootPart")

    if not Root then
        return
    end

    if not OriginalHitboxes[Player] then
        OriginalHitboxes[Player] = {
            Size = Root.Size,
            Transparency = Root.Transparency,
            CanCollide = Root.CanCollide
        }
    end

    Root.Size = Vector3.new(500, 500, 500)
    Root.Transparency = 0.8
    Root.CanCollide = false
end

local function RestoreHitbox(Player)
    local Character = Player.Character

    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local Original =
        OriginalHitboxes[Player]

    if Root and Original then
        Root.Size = Original.Size
        Root.Transparency = Original.Transparency
        Root.CanCollide = Original.CanCollide
    end

    OriginalHitboxes[Player] = nil
end

MainTab:CreateToggle({

        HitboxEnabled = Value

        if Value then
            for _, Player in ipairs(Players:GetPlayers()) do
                SetHitbox(Player)
            end
        else
            for _, Player in ipairs(Players:GetPlayers()) do
                RestoreHitbox(Player)
            end
        end
    end
})

local FlingPlayersEnabled = false
local FlingParts = {}

local function GetFlingTargets()
    local Targets = {}

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player.UserId ~= LocalPlayer.UserId then
            local Character = Player.Character
            local Humanoid =
                Character
                and Character:FindFirstChildOfClass("Humanoid")
            local Root =
                Character
                and Character:FindFirstChild("HumanoidRootPart")

            if Humanoid and Humanoid.Health > 0 and Root then
                table.insert(Targets, Root)
            end
        end
    end

    return Targets
end

local function GetUnanchoredParts()
    local Parts = {}
    local GameSystem = workspace:FindFirstChild("GameSystem", true)

    if not GameSystem then
        return Parts
    end

    local DeleteAfterMinigames = {}

    for _, Object in ipairs(GameSystem:GetDescendants()) do
        if Object.Name == "DeleteAfterMinigame" then
            table.insert(DeleteAfterMinigames, Object)
        end
    end

    local DirectDeleteAfterMinigame =
        GameSystem:FindFirstChild("DeleteAfterMinigame")

    if DirectDeleteAfterMinigame then
        table.insert(DeleteAfterMinigames, DirectDeleteAfterMinigame)
    end

    local Character = LocalPlayer.Character

    for _, DeleteAfterMinigame in ipairs(DeleteAfterMinigames) do
        for _, Object in ipairs(DeleteAfterMinigame:GetDescendants()) do
            if Object:IsA("BasePart")
                and not Object.Anchored
                and (not Character or not Object:IsDescendantOf(Character)) then
                table.insert(Parts, Object)
            end
        end
    end

    return Parts
end

local function FlingTarget(TargetRoot, Part)
    if not TargetRoot or not TargetRoot.Parent or not Part or not Part.Parent then
        return
    end

    if Part.Anchored then
        return
    end

    if not FlingParts[Part] then
        FlingParts[Part] = {
            CFrame = Part.CFrame,
            AssemblyLinearVelocity = Part.AssemblyLinearVelocity,
            AssemblyAngularVelocity = Part.AssemblyAngularVelocity
        }
    end

    pcall(function()
        Part.CFrame =
            TargetRoot.CFrame
            * CFrame.new(0, 0, -2)

        Part.AssemblyLinearVelocity =
            TargetRoot.AssemblyLinearVelocity
            + Vector3.new(0, 250, 0)

        Part.AssemblyAngularVelocity =
            Vector3.new(0, 10000, 0)
    end)
end

local function RestoreFlingParts()
    for Part, Original in pairs(FlingParts) do
        if Part and Part.Parent and not Part.Anchored then
            pcall(function()
                Part.CFrame = Original.CFrame
                Part.AssemblyLinearVelocity = Original.AssemblyLinearVelocity
                Part.AssemblyAngularVelocity = Original.AssemblyAngularVelocity
            end)
        end
    end

    table.clear(FlingParts)
end

TrollTab:CreateToggle({
    Name = "Fling Players",
    CurrentValue = false,

    Callback = function(Value)
        FlingPlayersEnabled = Value

        if not Value then
            RestoreFlingParts()
        end
    end
})

task.spawn(function()
    while task.wait(0.05) do
        if FlingPlayersEnabled then
            local Targets = GetFlingTargets()
            local Parts = GetUnanchoredParts()

            for _, TargetRoot in ipairs(Targets) do
                for _, Part in ipairs(Parts) do
                    if not FlingPlayersEnabled then
                        break
                    end

                    FlingTarget(TargetRoot, Part)
                end
            end
        end
    end
end)
