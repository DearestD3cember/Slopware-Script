local ESPEnabled = false
local ESPObjects = {}

local function RemoveESP(Player)
    if ESPObjects[Player] then
        for _, Object in ipairs(ESPObjects[Player]) do
            if Object and Object.Parent then
                Object:Destroy()
            end
        end
        ESPObjects[Player] = nil
    end
end

local function AddESP(Player)
    if not Player or Player.Name ~= "CriminalBlueGuy" then
        return
    end

    RemoveESP(Player)

    local Character = Player.Character
    if not Character then
        return
    end

    local Highlight = Instance.new("Highlight")
    Highlight.Name = "SlopwareESP"
    Highlight.Adornee = Character
    Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    Highlight.FillTransparency = 0.5
    Highlight.OutlineTransparency = 0
    Highlight.Parent = Character

    local Billboard = Instance.new("BillboardGui")
    Billboard.Name = "SlopwareESPText"
    Billboard.Size = UDim2.fromOffset(150, 40)
    Billboard.StudsOffset = Vector3.new(0, 3.5, 0)
    Billboard.AlwaysOnTop = true
    Billboard.Parent = Character

    local Text = Instance.new("TextLabel")
    Text.Size = UDim2.fromScale(1, 1)
    Text.BackgroundTransparency = 1
    Text.Text = "right here"
    Text.TextScaled = true
    Text.Font = Enum.Font.GothamBold
    Text.TextColor3 = Color3.new(1, 1, 1)
    Text.TextStrokeTransparency = 0
    Text.Parent = Billboard

    ESPObjects[Player] = {
        Highlight,
        Billboard
    }
end

local function WatchESPPlayer(Player)
    Player.CharacterAdded:Connect(function()
        task.wait(0.5)

        if ESPEnabled then
            AddESP(Player)
        end
    end)

    if ESPEnabled and Player.Character then
        AddESP(Player)
    end
end

local function GardenPartIsOwnedByLocalPlayer(GardenPart)
    local Owner = GardenPart:FindFirstChild("PartOwner", true)

    if not Owner then
        return false
    end

    local OwnerValue

    pcall(function()
        OwnerValue = Owner.Value
    end)

    if typeof(OwnerValue) == "Instance" then
        return OwnerValue == LocalPlayer
    end

    if typeof(OwnerValue) == "number" then
        return OwnerValue == LocalPlayer.UserId
    end

    if typeof(OwnerValue) == "string" then
        return OwnerValue == LocalPlayer.Name
            or OwnerValue == LocalPlayer.DisplayName
            or OwnerValue == tostring(LocalPlayer.UserId)
    end

    return false
end

local AutoShoot = false
local CurrentTarget = nil
local TargetStartTime = 0
local LastTarget = nil

local function GetRandomTarget()
    local Targets = {}

    for _, Player in ipairs(Players:GetPlayers()) do
        if Player ~= LocalPlayer and Player ~= LastTarget then
            local Character = Player.Character
            local Humanoid =
                Character
                and Character:FindFirstChildOfClass("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                table.insert(Targets, Player)
            end
        end
    end

    if #Targets == 0 then
        for _, Player in ipairs(Players:GetPlayers()) do
            if Player ~= LocalPlayer then
                local Character = Player.Character
                local Humanoid =
                    Character
                    and Character:FindFirstChildOfClass("Humanoid")

                if Humanoid and Humanoid.Health > 0 then
                    table.insert(Targets, Player)
                end
            end
        end
    end

    if #Targets > 0 then
        return Targets[math.random(1, #Targets)]
    end

    return nil
end

local function ShootTarget(TargetPlayer)
    if not TargetPlayer then
        return false
    end

    local Character = LocalPlayer.Character

    if not Character then
        return false
    end

    local Root =
        Character:FindFirstChild("HumanoidRootPart")

    local Pistol =
        Character:FindFirstChild("PistolTool")

    local Activation =
        Pistol
        and Pistol:FindFirstChild("Activation")

    if not Root or not Activation then
        return false
    end

    local TargetCharacter = TargetPlayer.Character

    if not TargetCharacter then
        return false
    end

    local TargetRoot =
        TargetCharacter:FindFirstChild("HumanoidRootPart")

    local Humanoid =
        TargetCharacter:FindFirstChildOfClass("Humanoid")

    if not TargetRoot
        or not Humanoid
        or Humanoid.Health <= 0 then
        return false
    end

    Root.CFrame =
        TargetRoot.CFrame
        * CFrame.new(0, 0, 5)

    pcall(function()
        Activation:FireServer(
            TargetRoot,
            TargetRoot.CFrame
        )
    end)

    return true
end

local function AutoWinShoot()
    if not CurrentTarget
        or not CurrentTarget.Character
        or not CurrentTarget.Character:FindFirstChildOfClass("Humanoid")
        or CurrentTarget.Character:FindFirstChildOfClass("Humanoid").Health <= 0
        or tick() - TargetStartTime >= 2 then

        LastTarget = CurrentTarget
        CurrentTarget = GetRandomTarget()
        TargetStartTime = tick()
    end

    if CurrentTarget then
        ShootTarget(CurrentTarget)
    end
end

task.spawn(function()
    while task.wait(0.1) do
        if AutoShoot and not AutoDetectWin then
            local TargetDead = false

            if CurrentTarget then
                local Character = CurrentTarget.Character
                local Humanoid =
                    Character
                    and Character:FindFirstChildOfClass("Humanoid")

                if not Humanoid or Humanoid.Health <= 0 then
                    TargetDead = true
                end
            end

            if not CurrentTarget
                or TargetDead
                or tick() - TargetStartTime >= 2 then

                LastTarget = CurrentTarget
                CurrentTarget = GetRandomTarget()
                TargetStartTime = tick()
            end

            if CurrentTarget then
                ShootTarget(CurrentTarget)
            end
        elseif not AutoShoot then
            CurrentTarget = nil
            LastTarget = nil
            TargetStartTime = 0
        end
    end
end)

local function GetLocalPlayerMap()
    local GameSystem =
        workspace:FindFirstChild("GameSystem", true)

    if not GameSystem then
        return nil
    end

    local DeleteAfterMinigames = {}

    for _, Object in ipairs(GameSystem:GetDescendants()) do
        if Object.Name == "DeleteAfterMinigame" then
            table.insert(DeleteAfterMinigames, Object)
        end
    end

    local DirectDeleteAfterMinigame =
        GameSystem:FindFirstChild("DeleteAfterMinigame")

    if DirectDeleteAfterMinigame then
        table.insert(DeleteAfterMinigames, DirectDeleteAfterMinigame)
    end

    local ExpectedName = LocalPlayer.Name .. "Map"

    for _, DeleteAfterMinigame in ipairs(DeleteAfterMinigames) do
        for _, Object in ipairs(DeleteAfterMinigame:GetDescendants()) do
            if Object.Name == "Map" then
                local PlayerMap =
                    Object:FindFirstChild(ExpectedName)

                if PlayerMap then
                    return PlayerMap
                end
            end

            if Object.Name == ExpectedName then
                return Object
            end
        end
    end

    return nil
end

local function GetObjectCFrame(Object)
    if not Object or not Object.Parent then
        return nil
    end

    if Object:IsA("BasePart") then
        return Object.CFrame
    end

    if Object:IsA("Model") then
        return Object:GetPivot()
    end

    if Object:IsA("Attachment") then
        return Object.WorldCFrame
    end

    local Part = Object:FindFirstChildWhichIsA("BasePart", true)

    if Part then
        return Part.CFrame
    end

    return nil
end

local function TeleportToObject(Object)
    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local TargetCFrame = GetObjectCFrame(Object)

    if not Root or not TargetCFrame then
        return false
    end

    pcall(function()
        Root.CFrame =
            TargetCFrame + Vector3.new(0, 3, 0)
    end)

    return true
end

local function ExpandEnemyHitbox(Enemy)
    if not Enemy or not Enemy.Parent then
        return
    end

    if not EnemyNames[Enemy.Name] then
        return
    end

    local Root =
        Enemy:FindFirstChild("HumanoidRootPart", true)

    if Root and Root:IsA("BasePart") then
        pcall(function()
            Root.Size = EnemyHitboxSize
            Root.Transparency = 0.8
            Root.CanCollide = false
        end)
    end
end

local function WatchEnemyFolder(Enemies)
    if not Enemies then
        return
    end

    for _, Enemy in ipairs(Enemies:GetChildren()) do
        ExpandEnemyHitbox(Enemy)
    end

    if WatchedEnemyFolders[Enemies] then
        return
    end

    WatchedEnemyFolders[Enemies] = true

    Enemies.ChildAdded:Connect(function(Enemy)
        task.wait()

        if AutoDetectWin then
            ExpandEnemyHitbox(Enemy)
        end
    end)
end

local function GetSwordEvent()
    local Character = LocalPlayer.Character

    if not Character then
        return nil
    end

    return Character:FindFirstChild("UseSword")
end

local function AttackEnemy(Enemy)
    if not Enemy or not Enemy.Parent then
        return false
    end

    if not EnemyNames[Enemy.Name] then
        return false
    end

    local Character = LocalPlayer.Character
    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    if not Root then
        return false
    end

    local EnemyRoot =
        Enemy:FindFirstChild("HumanoidRootPart", true)

    if not EnemyRoot and Enemy:IsA("BasePart") then
        EnemyRoot = Enemy
    end

    if not EnemyRoot then
        return false
    end

    ExpandEnemyHitbox(Enemy)

    pcall(function()
        -- Stay underneath the enemy to avoid damage.
        Root.CFrame =
            EnemyRoot.CFrame
            * CFrame.new(0, -15, 0)
    end)

    local Event = GetSwordEvent()

    if not Event then
        return false
    end

    pcall(function()
        Event:FireServer()
    end)

    return true
end

local function GetTileNumber(Tile)
    local Number =
        string.match(Tile.Name, "^Tile(%d+)$")

    return tonumber(Number) or math.huge
end

local function GetTiles(UserMap)
    local Tiles = {}

    for _, Object in ipairs(UserMap:GetChildren()) do
        if Object:IsA("Model")
            or Object:IsA("Folder") then

            if string.match(Object.Name, "^Tile%d+$") then
                table.insert(Tiles, Object)
            end
        end
    end

    table.sort(Tiles, function(A, B)
        local ANumber = GetTileNumber(A)
        local BNumber = GetTileNumber(B)

        if ANumber == BNumber then
            return A.Name < B.Name
        end

        return ANumber < BNumber
    end)

    return Tiles
end

local function GetOpenGate(UserMapOrTile)
    local StartPath =
        UserMapOrTile:FindFirstChild("StartPath", true)

    if StartPath then
        local Gate =
            StartPath:FindFirstChild("OpenGate", true)

        if Gate then
            return Gate
        end
    end

    return UserMapOrTile:FindFirstChild("OpenGate", true)
end

local function GetEnemyFolder(Tile)
    return Tile:FindFirstChild("Enemies", true)
end

local function HasLivingEnemy(Enemies)
    if not Enemies then
        return false
    end

    for _, Enemy in ipairs(Enemies:GetChildren()) do
        if EnemyNames[Enemy.Name] then
            local Humanoid =
                Enemy:FindFirstChildOfClass("Humanoid")
                or Enemy:FindFirstChild("Humanoid", true)

            if not Humanoid or Humanoid.Health > 0 then
                return true
            end
        end
    end

    return false
end

local function GetEnemiesLeftValue(Tile)
    local EnemiesLeft =
        Tile:FindFirstChild("EnemiesLeft", true)

    if not EnemiesLeft then
        return nil
    end

    local Value

    pcall(function()
        Value = EnemiesLeft.Value
    end)

    if typeof(Value) == "number" then
        return Value
    end

    return nil
end

local function HandleTile(Tile)
    if not Tile or not Tile.Parent then
        return false
    end

    local Gate = GetOpenGate(Tile)

    if Gate and Gate ~= LastPlayerMapGate then
        if TeleportToObject(Gate) then
            LastPlayerMapGate = Gate
            task.wait(0.15)
        end
    end

    local Enemies = GetEnemyFolder(Tile)

    if not Enemies then
        return true
    end

    WatchEnemyFolder(Enemies)

    local EmptyChecks = 0

    while AutoDetectWin and Tile.Parent do
        local EnemiesLeft = GetEnemiesLeftValue(Tile)
        local FoundEnemy = false

        for _, Enemy in ipairs(Enemies:GetChildren()) do
            if EnemyNames[Enemy.Name] then
                local Humanoid =
                    Enemy:FindFirstChildOfClass("Humanoid")
                    or Enemy:FindFirstChild("Humanoid", true)

                if not Humanoid or Humanoid.Health > 0 then
                    FoundEnemy = true
                    ExpandEnemyHitbox(Enemy)
                    AttackEnemy(Enemy)
                    task.wait(0.1)

                    if not AutoDetectWin then
                        return false
                    end
                end
            end
        end

        if not FoundEnemy then
            if typeof(EnemiesLeft) == "number"
                and EnemiesLeft > 0 then
                EmptyChecks = 0
                task.wait(0.1)
            else
                EmptyChecks += 1

                if EmptyChecks >= 3 then
                    break
                end

                task.wait(0.1)
            end
        else
            EmptyChecks = 0
        end
    end

    return true
end

local function AutoWinPlayerMap()
    local UserMap = GetLocalPlayerMap()

    if not UserMap then
        return false
    end

    if not PlayerMapStarted then
        local StartPath =
            UserMap:FindFirstChild("StartPath", true)

        local OpenGate =
            StartPath
            and StartPath:FindFirstChild("OpenGate", true)

        if not OpenGate then
            OpenGate =
                UserMap:FindFirstChild("OpenGate", true)
        end

        if OpenGate and TeleportToObject(OpenGate) then
            PlayerMapStarted = true
            LastPlayerMapGate = OpenGate
            task.wait(0.25)
        end
    end

    local ProcessedTiles = {}

    while AutoDetectWin and UserMap.Parent do
        local Tiles = GetTiles(UserMap)
        local DidWork = false

        for _, Tile in ipairs(Tiles) do
            if not AutoDetectWin then
                break
            end

            if not ProcessedTiles[Tile] then
                ProcessedTiles[Tile] = true
                LastPlayerMapTile = Tile

                local Gate = GetOpenGate(Tile)

                if Gate then
                    TeleportToObject(Gate)
                    LastPlayerMapGate = Gate
                    task.wait(0.15)
                end

                HandleTile(Tile)
                DidWork = true
            end
        end

        if not DidWork then
            task.wait(0.2)
        end
    end

    return true
end

local HitboxEnabled = false
local OriginalHitboxes = {}

local function SetHitbox(Player)
    if Player == LocalPlayer then
        return
    end

    local Character = Player.Character

    if not Character then
        return
    end

    local Root =
        Character:FindFirstChild("HumanoidRootPart")

    if not Root then
        return
    end

    if not OriginalHitboxes[Player] then
        OriginalHitboxes[Player] = {
            Size = Root.Size,
            Transparency = Root.Transparency,
            CanCollide = Root.CanCollide
        }
    end

    Root.Size = Vector3.new(100, 100, 100)
    Root.Transparency = 0.8
    Root.CanCollide = false
end

local function RestoreHitbox(Player)
    local Character = Player.Character

    local Root =
        Character
        and Character:FindFirstChild("HumanoidRootPart")

    local Original =
        OriginalHitboxes[Player]

    if Root and Original then
        Root.Size = Original.Size
        Root.Transparency = Original.Transparency
        Root.CanCollide = Original.CanCollide
    end

    OriginalHitboxes[Player] = nil
end

MainTab:CreateToggle({
    Name = "Hitbox Expander",
    CurrentValue = false,

    Callback = function(Value)
        HitboxEnabled = Value

        if Value then
            for _, Player in ipairs(Players:GetPlayers()) do
                SetHitbox(Player)
            end
        else
            for _, Player in ipairs(Players:GetPlayers()) do
                RestoreHitbox(Player)
            end
        end
    end
})

local function SetupHitboxPlayer(Player)
    if Player == LocalPlayer then
        return
    end

    Player.CharacterAdded:Connect(function()
        task.wait(1)

        if HitboxEnabled then
            SetHitbox(Player)
        end
    end)
end

for _, Player in ipairs(Players:GetPlayers()) do
    SetupHitboxPlayer(Player)
end

Players.PlayerAdded:Connect(function(Player)
    SetupHitboxPlayer(Player)
end)

Players.PlayerRemoving:Connect(function(Player)
    OriginalHitboxes[Player] = nil
end)

local Baseplate

MainTab:CreateToggle({
    Name = "Baseplate to Survive",
    CurrentValue = false,

    Callback = function(Value)
        if Value then
            local Character = LocalPlayer.Character

            local Root =
                Character
                and Character:FindFirstChild("HumanoidRootPart")

            if not Root then
                return
            end

            if Baseplate then
                Baseplate:Destroy()
            end

            Baseplate = Instance.new("Part")
            Baseplate.Name = "Baseplate"
            Baseplate.Size = Vector3.new(500, 2, 500)
            Baseplate.Position =
                Root.Position + Vector3.new(0, 1000, 0)

            Baseplate.Anchored = true
            Baseplate.CanCollide = true
            Baseplate.CanTouch = true
            Baseplate.CanQuery = true
            Baseplate.Material = Enum.Material.SmoothPlastic
            Baseplate.Parent = workspace

            Root.CFrame =
                Baseplate.CFrame + Vector3.new(0, 5, 0)
        else
            if Baseplate then
                Baseplate:Destroy()
                Baseplate = nil
            end
        end
    end
})

local NoclipEnabled = false
local NoclipOriginals = {}

local function ApplyNoclip()
    local Character = LocalPlayer.Character

    if not Character then
        return
    end

    for _, Object in ipairs(Character:GetDescendants()) do
        if Object:IsA("BasePart") then
            if NoclipOriginals[Object] == nil then
                NoclipOriginals[Object] = Object.CanCollide
            end

            Object.CanCollide = false
        end
    end
end

local function RestoreNoclip()
    for Object, Original in pairs(NoclipOriginals) do
        if Object and Object.Parent then
            Object.CanCollide = Original
        end
    end

    table.clear(NoclipOriginals)
end

MainTab:CreateToggle({
    Name = "Noclip",
    CurrentValue = false,

    Callback = function(Value)
        NoclipEnabled = Value

        if not Value then
            RestoreNoclip()
        end
    end
})

task.spawn(function()
    while task.wait(0.1) do
        if NoclipEnabled then
            ApplyNoclip()
        end
    end
end)

local MoreSpeed = false
local SpeedValue = 50
local OriginalWalkSpeed = {}

local function ApplySpeed(Character)
    local Humanoid =
        Character
        and Character:FindFirstChildOfClass("Humanoid")

    if not Humanoid then
        return
    end

    if OriginalWalkSpeed[Humanoid] == nil then
        OriginalWalkSpeed[Humanoid] = Humanoid.WalkSpeed
    end

    Humanoid.WalkSpeed = SpeedValue
end

local function RestoreSpeed()
    for Humanoid, Speed in pairs(OriginalWalkSpeed) do
        if Humanoid and Humanoid.Parent then
            pcall(function()
                Humanoid.WalkSpeed = Speed
            end)
        end
    end

    table.clear(OriginalWalkSpeed)
end

MainTab:CreateToggle({
    Name = "Get Faster",
    CurrentValue = false,

    Callback = function(Value)
        MoreSpeed = Value

        if not Value then
            RestoreSpeed()
        end
    end
})

task.spawn(function()
    while task.wait(0.1) do
        if MoreSpeed then
            local Character = LocalPlayer.Character

            if Character then
                ApplySpeed(Character)
            end
        end
    end
end)

LocalPlayer.CharacterAdded:Connect(function(Character)
    task.wait(0.5)

    if NoclipEnabled then
        ApplyNoclip()
    end

    if MoreSpeed then
        ApplySpeed(Character)
    end
end)
